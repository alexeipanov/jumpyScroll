Basic usage

Include CSS and JS


<link rel="stylesheet" href="jumpyscroll.css">
<script src="jumpyscroll.min.js"></script>

HTML


<body>
    <section>
        This is a first section!
    </section>
    <section>
        ...
    </section>
</body>

JS

$(document).ready(function() {  
    window.jumpyScroll();  
});

Next: events

Events

JS

window.jumpyScroll({
    onAfterScroll: function(index) {
        switch (index) {
            case 0:
                jumpyScroll.settings.infoText = 'This is first page!';
                break;
            case 1:
                post('seekTo', 54);
                post('play');
                break;
        }
    },
    onBeforeScroll: function(index) {
        switch (index) {
            default:
                post('pause');
                jumpyScroll.settings.infoText = 'Page ${pageNo} from ${pageCount}';
                break;
        }
    }
});
